  return (
    <Suspense fallback={<div className="min-h-screen bg-brand-bg flex items-center justify-center font-pixel text-brand-text">Đang tải...</div>}>

function CardsPageContent() {
  const { addPoints } = useApp();
  const searchParams = useSearchParams();
  const tab = searchParams.get("tab");

  // Game configuration states
  const [gameState, setGameState] = useState<"setup" | "playing" | "summary">("setup");
  const [gameMode, setGameMode] = useState<"level" | "free">("level");
  const [currentLevel, setCurrentLevel] = useState<number>(1); // For level mode, level 1 to 4
  const [showGuideModal, setShowGuideModal] = useState(false);

  const activePlayerRef = React.useRef<HTMLDivElement>(null);
  const pageTopRef = React.useRef<HTMLDivElement>(null);
  
  // Players list state
  const [players, setPlayers] = useState<Player[]>([
    { name: "Bố", role: "parent", score: 0 },
    { name: "Mẹ", role: "parent", score: 0 },
    { name: "Vy", role: "child", score: 0 }
  ]);
  const [newPlayerName, setNewPlayerName] = useState("");
  const [newPlayerRole, setNewPlayerRole] = useState<Player["role"]>("child");
  const [currentPlayerIndex, setCurrentPlayerIndex] = useState(0);

  // Auto-scroll active player into center of horizontal queue
  useEffect(() => {
    if (gameState === "playing" && activePlayerRef.current) {
      activePlayerRef.current.scrollIntoView({
        behavior: "smooth",
        block: "nearest",
        inline: "center"
      });
    }
  }, [currentPlayerIndex, gameState]);

  // Play session states
  const [currentStep, setCurrentStep] = useState<1 | 2 | 3>(1);
  const [activeDeckKey, setActiveDeckKey] = useState<string | null>(null);
  const [currentCard, setCurrentCard] = useState<Card | null>(null);
  const [isFlipped, setIsFlipped] = useState(true);
  const [criteriaChecked, setCriteriaChecked] = useState<boolean[]>([]);
  const [isSafeActivated, setIsSafeActivated] = useState(false);
  const [historyLog, setHistoryLog] = useState<string[]>([]);

  // Calculate dynamic card count suitable for current players roles
  const getSuitableCards = (playerList: Player[]) => {
    const hasParent = playerList.some(p => p.role === "parent");
    const hasChild = playerList.some(p => p.role === "child");
    const hasGrandparent = playerList.some(p => p.role === "grandparent");
    const childCount = playerList.filter(p => p.role === "child").length;

    return CARDS_DATA.filter((card) => {
      if (card.symbol === "BM-C") {
        return hasParent && hasChild;
      }
      if (card.symbol === "ACE") {
        return childCount >= 2;
      }
      if (card.symbol === "OB-C") {
        return hasGrandparent && hasChild;
      }
      if (card.symbol === "TH") {
        return (hasParent || hasGrandparent) && hasChild;
      }
      return true; // GD, BT, AT always suitable
    });
  };

  const suitableCards = getSuitableCards(players);
  const activeDeck = DECKS_CONFIG.find((d) => d.key === activeDeckKey);

  // Automatically open card draw when tab=today query parameter is set
  useEffect(() => {
    if (tab === "today") {
      setGameState("playing");
      setGameMode("free");
      const randomDecks = ["HEART", "WARM", "PLAY", "BOND"];
      const randomDeck = randomDecks[Math.floor(Math.random() * randomDecks.length)];
      drawCardForCategory(randomDeck);
    }
  }, [tab]);

  const addPlayer = () => {
    if (!newPlayerName.trim()) return;
    setPlayers([...players, { name: newPlayerName.trim(), role: newPlayerRole, score: 0 }]);
    setNewPlayerName("");
  };

  const removePlayer = (index: number) => {
    if (players.length <= 2) {
      alert("Cần có tối thiểu 2 người tham gia chơi!");
      return;
    }
    setPlayers(players.filter((_, i) => i !== index));
  };

  const startNewGame = () => {
    if (players.length < 2) {
      alert("Hãy thêm ít nhất 2 thành viên chơi để bắt đầu!");
      return;
    }
    const initialSuitable = getSuitableCards(players);
    if (initialSuitable.length === 0) {
      alert("Bộ bài phù hợp rỗng! Vui lòng cấu hình vai trò người chơi lại.");
      return;
    }
    setGameState("playing");
    setCurrentPlayerIndex(0);
    setHistoryLog([]);
    startPlayerTurn(0, gameMode === "level" ? "WARM" : null);
  };

  const getDeckKeyByLevelNumber = (lvl: number) => {
    switch (lvl) {
      case 1: return "WARM";
      case 2: return "PLAY";
      case 3: return "BOND";
      case 4: return "HEART";
      default: return "WARM";
    }
  };

  const startPlayerTurn = (playerIdx: number, forceDeckKey: string | null = null) => {
    setCurrentStep(1);
    setIsFlipped(true);
    setIsSafeActivated(false);

    let deckKey = forceDeckKey;
    if (!deckKey) {
      if (gameMode === "level") {
        deckKey = getDeckKeyByLevelNumber(currentLevel);
      } else {
        // Free Mode: draw from level 1 to level 4 randomly (exclude Safe level since it is triggered manually)
        const randomLevels = ["WARM", "PLAY", "BOND", "HEART"];
        deckKey = randomLevels[Math.floor(Math.random() * randomLevels.length)];
      }
    }

    drawCardForCategory(deckKey);
  };

  const drawCardForCategory = (deckKey: string) => {
    setActiveDeckKey(deckKey);
    const pool = suitableCards.filter(c => c.category === deckKey);
    if (pool.length === 0) {
      // Fallback to absolute pool if dynamic pool has no cards in this category
      const absolutePool = CARDS_DATA.filter(c => c.category === deckKey);
      const initialCard = absolutePool[Math.floor(Math.random() * absolutePool.length)];
      setCurrentCard(initialCard);
    } else {
      const initialCard = pool[Math.floor(Math.random() * pool.length)];
      setCurrentCard(initialCard);
    }
    
    // Reset criteria checkboxes
    const criteriaCount = SCORING_CRITERIA[deckKey]?.length || 0;
    setCriteriaChecked(new Array(criteriaCount).fill(true));
  };

  const handleCardFlip = () => {
    if (currentStep === 1) {
      setIsFlipped(false);
      setCurrentStep(2);
    }
  };

  // Convert current card into an emotional safety card
  const handleActivateSafeCard = () => {
    setIsSafeActivated(true);
    const safePool = suitableCards.filter(c => c.category === "SAFE");
    const safeCard = safePool.length > 0 
      ? safePool[Math.floor(Math.random() * safePool.length)]
      : CARDS_DATA.filter(c => c.category === "SAFE")[0];
    
    setCurrentCard(safeCard);
    setActiveDeckKey("SAFE");
    setCriteriaChecked([true]); // Auto-check the single safety score
    setCurrentStep(3); // Go straight to feedback/scoring
  };

  const handleFinishAnswering = () => {
    setCurrentStep(3);
  };

  const handleToggleCriterion = (index: number) => {
    const updated = [...criteriaChecked];
    updated[index] = !updated[index];
    setCriteriaChecked(updated);
  };

  const handleConfirmTurnScore = () => {
    if (!currentCard || !activeDeckKey) return;

    // Calculate score
    const criteriaList = SCORING_CRITERIA[activeDeckKey] || [];
    let turnPoints = 0;
    criteriaChecked.forEach((checked, idx) => {
      if (checked) {
        turnPoints += criteriaList[idx]?.pts || 0;
      }
    });

    // Award score to current player
    const updatedPlayers = [...players];
    updatedPlayers[currentPlayerIndex].score += turnPoints;
    setPlayers(updatedPlayers);

    // Add to global AppContext points
    addPoints(turnPoints);

    // Save to local logs
    const activePlayer = players[currentPlayerIndex];
    const deckInfo = DECKS_CONFIG.find(d => d.key === activeDeckKey);
    const logMessage = `${activePlayer.name} đã hoàn thành thẻ ${deckInfo?.subtitle} (Ký hiệu ${currentCard.symbol}), đạt +${turnPoints} điểm.`;
    setHistoryLog([logMessage, ...historyLog]);

    // confetti check for top points!
    if (turnPoints >= 3) {
      confetti({
        particleCount: 80,
        spread: 60,
        colors: ["#8B5CF6", "#F472B6", "#FBBF24"],
        origin: { y: 0.6 }
      });
    }

    // Determine next turn / next level
    let nextPlayerIdx = currentPlayerIndex + 1;
    let nextLevel = currentLevel;

    if (nextPlayerIdx >= players.length) {
      nextPlayerIdx = 0; // Wrap around to first player
      
      // If playing level mode, check if we should advance to the next level
      if (gameMode === "level") {
        if (currentLevel < 4) {
          nextLevel = currentLevel + 1;
          setCurrentLevel(nextLevel);
          alert(`Cả nhà đã hoàn thành xong Level ${currentLevel}! Chuyển sang Level ${nextLevel}: ${DECKS_CONFIG.find(d => d.key === getDeckKeyByLevelNumber(nextLevel))?.subtitle}`);
        } else {
          // Finished level 4! Go to summary
          setGameState("summary");
          return;
        }
      }
    }

    setCurrentPlayerIndex(nextPlayerIdx);
    startPlayerTurn(nextPlayerIdx, gameMode === "level" ? getDeckKeyByLevelNumber(nextLevel) : null);
  };

  const symbolInfo = currentCard ? SYMBOL_DETAILS[currentCard.symbol] : null;

  return (
    <div ref={pageTopRef} className="max-w-4xl mx-auto space-y-6 pb-12">
      
      {/* HEADER BAR */}
      <section className="bg-brand-card border-3 border-brand-outline p-6 rounded-3xl shadow-pixel">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
          <h2 className="font-pixel text-xl sm:text-2xl font-bold text-brand-text flex items-center gap-2">
            <i className="hn hn-grid-solid text-brand-purple text-[22px]" />
            80others - Game Kết Nối Gia Đình
          </h2>
          <div className="flex items-center gap-2 shrink-0">
            <button
              onClick={() => setShowGuideModal(true)}
              className="py-1.5 px-3 border-2 border-brand-outline bg-amber-100 hover:bg-amber-200 text-amber-700 font-pixel font-bold text-xs rounded-xl shadow-pixel-sm active:translate-y-0.5 cursor-pointer flex items-center gap-1"
            >
              <i className="hn hn-lightbulb-solid text-sm" />
              Hướng dẫn
            </button>
            {gameState === "playing" && (
              <button
                onClick={() => setGameState("summary")}
                className="py-1.5 px-3 border-2 border-brand-outline bg-red-100 hover:bg-red-200 text-red-700 font-pixel font-bold text-xs rounded-xl shadow-pixel-sm active:translate-y-0.5 cursor-pointer"
              >
                Dừng chơi
              </button>
            )}
          </div>
        </div>
        <p className="text-brand-text/80 text-xs sm:text-sm font-semibold mt-2">
          Bộ bài chia sẻ chân thành, lắng nghe thấu hiểu không phán xét, kết nối yêu thương gia đình Việt Nam.
        </p>
      </section>

      {/* -------------------- 1. SCREEN SETUP -------------------- */}
      {gameState === "setup" && (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6 animate-in fade-in duration-200">
          
          {/* Players configuration */}
          <div className="bg-brand-card border-3 border-brand-outline p-5 rounded-3xl shadow-pixel space-y-4">
            <h3 className="font-pixel text-base font-bold text-brand-text flex items-center gap-2 border-b-2 border-brand-border pb-3">
              <i className="hn hn-users-solid text-brand-purple" />
              1. Thành viên tham gia
            </h3>
            
            <div className="space-y-2 max-h-60 overflow-y-auto pr-1">
              {players.map((p, idx) => (
                <div key={idx} className="flex items-center justify-between p-3 border-2 border-brand-outline rounded-2xl bg-brand-bg shadow-pixel-sm">
                  <div className="flex items-center gap-3">
                    <div className="w-8 h-8 rounded-full border border-brand-outline bg-brand-purple/20 flex items-center justify-center">
                      {p.role === "parent" ? (
                        <i className="hn hn-user-solid text-brand-purple text-sm" />
                      ) : p.role === "grandparent" ? (
                        <i className="hn hn-crown-solid text-brand-yellow text-sm" />
                      ) : (
                        <i className="hn hn-heart-solid text-brand-pink text-xs" />
                      )}
                    </div>
                    <div>
                      <span className="font-cozy font-bold text-sm text-brand-text">{p.name}</span>
                      <span className="text-[10px] text-brand-text/60 font-semibold block uppercase">
                        {p.role === "parent" ? "Bố / Mẹ" : p.role === "grandparent" ? "Ông / Bà" : "Con cái"}
                      </span>
                    </div>
                  </div>
                  <button 
                    onClick={() => removePlayer(idx)}
                    className="p-1 border border-brand-outline rounded-lg bg-red-50 text-red-500 hover:bg-red-100 cursor-pointer"
                  >
                    <i className="hn hn-times-solid text-xs" />
                  </button>
                </div>
              ))}
            </div>

            {/* Input field to add player */}
            <div className="flex flex-col sm:flex-row gap-2 pt-2 border-t border-brand-border">
              <input
                type="text"
                placeholder="Tên thành viên..."
                value={newPlayerName}
                onChange={(e) => setNewPlayerName(e.target.value)}
                className="flex-grow p-2.5 border-2 border-brand-outline rounded-xl font-cozy text-sm bg-brand-bg"
              />
              <div className="flex gap-2">
                <select
                  value={newPlayerRole}
                  onChange={(e) => setNewPlayerRole(e.target.value as Player["role"])}
                  className="p-2.5 border-2 border-brand-outline rounded-xl font-cozy text-xs bg-brand-bg font-semibold"
                >
                  <option value="child">Con cái</option>
                  <option value="parent">Bố / Mẹ</option>
                  <option value="grandparent">Ông / Bà</option>
                </select>
                <button
                  onClick={addPlayer}
                  className="py-2.5 px-4 border-2 border-brand-outline bg-brand-purple text-white font-pixel font-bold text-xs rounded-xl shadow-pixel-sm active:translate-y-0.5 cursor-pointer"
                >
                  Thêm
                </button>
              </div>
            </div>
          </div>

          {/* Game rules & settings */}
          <div className="bg-brand-card border-3 border-brand-outline p-5 rounded-3xl shadow-pixel flex flex-col justify-between space-y-5">
            <div className="space-y-4">
              <h3 className="font-pixel text-base font-bold text-brand-text flex items-center gap-2 border-b-2 border-brand-border pb-3">
                <i className="hn hn-settings-solid text-brand-purple" />
                2. Thiết lập trò chơi
              </h3>

              {/* Mode Selection */}
              <div className="space-y-2">
                <label className="font-pixel text-xs font-bold text-brand-text/70 block">CHẾ ĐỘ CHƠI</label>
                <div className="grid grid-cols-2 gap-3">
                  <div
                    onClick={() => setGameMode("level")}
                    className={`p-3 border-2 rounded-2xl cursor-pointer transition-all shadow-pixel-sm ${
                      gameMode === "level"
                        ? "border-brand-purple bg-brand-purple/10 text-brand-purple font-bold"
                        : "border-brand-outline bg-brand-bg text-brand-text/70"
                    }`}
                  >
                    <p className="font-pixel text-xs">Cấp độ (Level)</p>
                    <span className="text-[10px] block leading-tight mt-1">Lần lượt từ dễ tới sâu sắc (Level 1 đến 4)</span>
                  </div>
                  <div
                    onClick={() => setGameMode("free")}
                    className={`p-3 border-2 rounded-2xl cursor-pointer transition-all shadow-pixel-sm ${
                      gameMode === "free"
                        ? "border-brand-purple bg-brand-purple/10 text-brand-purple font-bold"
                        : "border-brand-outline bg-brand-bg text-brand-text/70"
                    }`}
                  >
                    <p className="font-pixel text-xs">Chơi Tự Do</p>
                    <span className="text-[10px] block leading-tight mt-1">Xáo tất cả lá bài và rút ngẫu nhiên</span>
                  </div>
                </div>
              </div>

              {/* Suitable Cards Info */}
              <div className="bg-brand-bg border-2 border-brand-border p-3.5 rounded-2xl text-xs space-y-2">
                <div className="flex items-center justify-between font-bold text-brand-text">
                  <span>Bộ bài tự động lọc:</span>
